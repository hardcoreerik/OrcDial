cmake_minimum_required(VERSION 3.20)
project(orcdial_security_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT MBEDTLS_SOURCE_DIR)
  message(FATAL_ERROR "Pass -DMBEDTLS_SOURCE_DIR=<mbedtls source tree>")
endif()
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
add_subdirectory("${MBEDTLS_SOURCE_DIR}" mbedtls EXCLUDE_FROM_ALL)
add_executable(security_test security_test.cpp)
target_link_libraries(security_test PRIVATE mbedcrypto)
add_executable(protocol_test protocol_test.cpp)
add_executable(controller_test controller_test.cpp)
add_executable(runtime_test runtime_test.cpp)
target_include_directories(runtime_test PRIVATE runtime_stubs)
target_link_libraries(runtime_test PRIVATE mbedcrypto)
enable_testing()
add_test(NAME runtime_failure COMMAND runtime_test)
add_test(NAME security COMMAND security_test)
add_test(NAME protocol COMMAND protocol_test)
add_test(NAME controller COMMAND controller_test)
